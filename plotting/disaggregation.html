地震危险性解聚

PSHA 得到危险性水平后，需要回答"这个危险性来自哪些地震场景"。 解聚分析把目标危险性分解为震级 M、距离 R、epsilon 三个变量的联合贡献。 页面提供 2D 四面板 与 3D 立体视图 两种展示方式。

工程地震 PSHA CSV 输入 2D / 3D A4 尺寸
CSV Input

CSV 输入数据

本页读取 assets/disaggregation.csv 作为输入。 文件是 PSHA 解聚结果的逐分箱输出，每行对应一个 (M, R, epsilon) 组合的贡献百分比。

CSV 格式说明

列名 物理量 单位 说明
M_center震级 M 分箱中心无量纲通常 0.25 或 0.5 分箱
R_center距离 R 分箱中心kmJoyner-Boore 距离
epsilonepsilon 分箱中心无量纲对数残差的标准分数
contribution贡献百分比%所有分箱之和 ≈ 100

Meaning

两种视图的用途

2D 四面板（信息最全）

(a) M-R 联合分布热图 · (b) M 边际分布 · (c) R 边际分布 · (d) ε 分布。

3D 立体视图（一图概览）

经典 matplotlib bar3d 视角：三面墙（后墙、右墙、底面）带网格， X 轴（震级）水平，Y 轴（距离）斜向右上，Z 轴（贡献率）竖直向上。 每个 (M, R) 格子中央一根柱子，柱高 = 贡献百分比。

用途与局限：解聚结果用于挑选和调整真实地震记录， 或构造条件均值谱（CMS）。但它给出的是"场景的统计特征"， 不能作为单次地震的代表。

disaggregation.pyPython
# =============================================================
# 地震危险性解聚 | M-R-epsilon 联合贡献
# -------------------------------------------------------------
# 输入数据：disaggregation.csv
#   M_center,R_center,epsilon,contribution
#   5.25,7.5,-0.5,0.15
#   5.25,7.5,0.5,0.27
#   ...
#   单位：M 无量纲, R km, epsilon 无量纲, contribution %
# -------------------------------------------------------------
# 输出：
#   disaggregation.png      2D 四面板（热图 + 3 个边际分布）
#   disaggregation-3d.png   3D 立体柱状图（经典 bar3d 视角）
# =============================================================

import os
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.gridspec import GridSpec
from mpl_toolkits.mplot3d import Axes3D  # noqa: F401

# ---------- 全局字体 ----------
plt.rcParams.update({
    'font.family': 'serif',
    'font.serif': ['Times New Roman', 'Songti SC', 'SimSun', 'DejaVu Sans'],
    'font.size': 9,
    'axes.linewidth': 0.8,
    'axes.unicode_minus': False,
    'xtick.direction': 'in',
    'ytick.direction': 'in',
})

CSV_FILE = 'disaggregation.csv'


# =============================================================
# 1. 读取与聚合
# =============================================================
def generate_synthetic():
    """CSV 缺失时的合成数据（与网页版内联数据一致）。"""
    Ms = [5.25, 5.75, 6.25, 6.75, 7.25]
    Rs = [7.5, 15.0, 25.0, 40.0]
    eps = [-0.5, 0.5, 1.5]
    fM = {5.25: 0.20, 5.75: 0.60, 6.25: 1.00, 6.75: 0.85, 7.25: 0.35}
    fR = {7.5: 0.30, 15.0: 1.00, 25.0: 0.85, 40.0: 0.45}
    fE = {-0.5: 0.20, 0.5: 0.35, 1.5: 0.45}

    M_all, R_all, E_all, C_all = [], [], [], []
    for M in Ms:
        for R in Rs:
            w = fM[M] * fR[R]
            for e in eps:
                M_all.append(M)
                R_all.append(R)
                E_all.append(e)
                C_all.append(w * fE[e])
    C_all = np.array(C_all)
    C_all = C_all / C_all.sum() * 100
    return (np.array(M_all), np.array(R_all),
            np.array(E_all), C_all)


def read_csv(path):
    """读取 CSV，跳过 # 注释行，返回 4 列数组。"""
    M_all, R_all, eps_all, c_all = [], [], [], []
    with open(path, 'r', encoding='utf-8') as f:
        for line in f:
            line = line.strip()
            if not line or line.startswith('#'):
                continue
            if line.lower().startswith('m_center'):
                continue
            parts = line.split(',')
            if len(parts) < 4:
                continue
            try:
                M = float(parts[0])
                R = float(parts[1])
                e = float(parts[2])
                c = float(parts[3])
            except ValueError:
                continue
            if c <= 0:
                continue
            M_all.append(M); R_all.append(R)
            eps_all.append(e); c_all.append(c)
    return (np.array(M_all), np.array(R_all),
            np.array(eps_all), np.array(c_all))


if os.path.exists(CSV_FILE):
    M_all, R_all, eps_all, c_all = read_csv(CSV_FILE)
    print(f'从 {CSV_FILE} 读取 {len(c_all)} 个分箱')
else:
    print(f'{CSV_FILE} 不存在，使用合成数据演示')
    M_all, R_all, eps_all, c_all = generate_synthetic()

# 归一化到 100%
if abs(c_all.sum() - 100) > 0.5:
    c_all = c_all / c_all.sum() * 100

# 唯一分箱
M_bins = np.unique(M_all)
R_bins = np.unique(R_all)
eps_bins = np.unique(eps_all)
nM, nR, nE = len(M_bins), len(R_bins), len(eps_bins)

# 联合矩阵 + 加权 ε
MR_joint = np.zeros((nM, nR))
MR_eps_sum = np.zeros((nM, nR))
for i, m in enumerate(M_bins):
    for j, r in enumerate(R_bins):
        mask = (M_all == m) & (R_all == r)
        MR_joint[i, j] = c_all[mask].sum()
        MR_eps_sum[i, j] = (c_all[mask] * eps_all[mask]).sum()

MR_eps_mean = np.where(
    MR_joint > 0,
    MR_eps_sum / np.maximum(MR_joint, 1e-12),
    0.0
)

# 边际分布
M_marg = MR_joint.sum(axis=1)
R_marg = MR_joint.sum(axis=0)
eps_marg = np.array([c_all[eps_all == e].sum() for e in eps_bins])

# 峰值与均值
iM_peak = int(np.argmax(M_marg))
iR_peak = int(np.argmax(R_marg))
iE_peak = int(np.argmax(eps_marg))

M_mean = (M_marg * M_bins).sum() / M_marg.sum()
R_mean = (R_marg * R_bins).sum() / R_marg.sum()
eps_mean = (eps_marg * eps_bins).sum() / eps_marg.sum()

print(f'M 峰值 = {M_bins[iM_peak]:.2f}, 贡献 {M_marg[iM_peak]:.1f}%')
print(f'R 峰值 = {R_bins[iR_peak]:.1f} km, 贡献 {R_marg[iR_peak]:.1f}%')
print(f'ε 峰值 = {eps_bins[iE_peak]:.2f}, 贡献 {eps_marg[iE_peak]:.1f}%')
print(f'平均 M = {M_mean:.2f}, R = {R_mean:.1f} km, ε = {eps_mean:.2f}')


# =============================================================
# 2. 2D 四面板
# =============================================================
fig = plt.figure(figsize=(7.2, 8.4))
gs = GridSpec(3, 2, figure=fig, height_ratios=[1.6, 1.0, 1.0],
              hspace=0.55, wspace=0.35)

# ----- (a) M-R 联合分布热图 -----
ax_heat = fig.add_subplot(gs[0, :])
pcm = ax_heat.pcolormesh(
    np.arange(nR + 1), np.arange(nM + 1), MR_joint,
    cmap='OrRd', edgecolors='white', linewidth=1.5
)

# 单元格数字
cmax = MR_joint.max()
for i in range(nM):
    for j in range(nR):
        val = MR_joint[i, j]
        if val > 0.5:
            color = '#222222' if val < cmax * 0.6 else 'white'
            ax_heat.text(j + 0.5, i + 0.5, f'{val:.1f}',
                         ha='center', va='center',
                         fontsize=8, color=color)

ax_heat.set_xticks(np.arange(nR) + 0.5)
ax_heat.set_xticklabels([f'{r:g}' for r in R_bins])
ax_heat.set_yticks(np.arange(nM) + 0.5)
ax_heat.set_yticklabels([f'{m:.2f}' for m in M_bins])
ax_heat.set_xlabel('Joyner-Boore distance  R (km)')
ax_heat.set_ylabel('Magnitude M')
ax_heat.invert_yaxis()
ax_heat.set_title('(a) M-R disaggregation  ·  contribution %', fontsize=10)
cbar = fig.colorbar(pcm, ax=ax_heat, fraction=0.035, pad=0.02)
cbar.set_label('Contribution (%)', fontsize=8)

# ----- (b) M 边际分布 -----
ax_M = fig.add_subplot(gs[1, 0])
bars_M = ax_M.bar(np.arange(nM), M_marg, width=0.7,
                  color='#b45309', edgecolor='#222222', linewidth=0.6)
bars_M[iM_peak].set_color('#7c2d12')
ax_M.axvline(iM_peak, color='#7c2d12', lw=0.8, ls=':', alpha=0.8)
ax_M.set_xticks(np.arange(nM))
ax_M.set_xticklabels([f'{m:.2f}' for m in M_bins], fontsize=8)
ax_M.set_xlabel('Magnitude M')
ax_M.set_ylabel('Contribution (%)')
ax_M.set_title('(b) Marginal M distribution', fontsize=10)
ax_M.grid(True, axis='y', ls=':', lw=0.4, alpha=0.6)
ax_M.text(0.98, 0.95,
          f'peak = {M_bins[iM_peak]:.2f}\nmean = {M_mean:.2f}',
          transform=ax_M.transAxes, ha='right', va='top', fontsize=8,
          color='#7c2d12',
          bbox=dict(boxstyle='round,pad=0.3', facecolor='white',
                    edgecolor='#cccccc', linewidth=0.5))

# ----- (c) R 边际分布 -----
ax_R = fig.add_subplot(gs[1, 1])
bars_R = ax_R.bar(np.arange(nR), R_marg, width=0.7,
                  color='#2563eb', edgecolor='#222222', linewidth=0.6)
bars_R[iR_peak].set_color('#1e3a8a')
ax_R.axvline(iR_peak, color='#1e3a8a', lw=0.8, ls=':', alpha=0.8)
ax_R.set_xticks(np.arange(nR))
ax_R.set_xticklabels([f'{r:g}' for r in R_bins], fontsize=8)
ax_R.set_xlabel('R (km)')
ax_R.set_ylabel('Contribution (%)')
ax_R.set_title('(c) Marginal R distribution', fontsize=10)
ax_R.grid(True, axis='y', ls=':', lw=0.4, alpha=0.6)
ax_R.text(0.98, 0.95,
          f'peak = {R_bins[iR_peak]:.1f}\nmean = {R_mean:.1f} km',

          bbox=dict(boxstyle='round,pad=0.3', facecolor='white',

# ----- (d) Epsilon 分布 -----
ax_eps = fig.add_subplot(gs[2, :])
bars_E = ax_eps.bar(np.arange(nE), eps_marg, width=0.5,
                    color='#15803d', edgecolor='#222222', linewidth=0.6)
bars_E[iE_peak].set_color('#14532d')
ax_eps.axvline(iE_peak, color='#14532d', lw=0.8, ls=':', alpha=0.8)
ax_eps.set_xticks(np.arange(nE))
ax_eps.set_xticklabels([f'{e:.1f}' for e in eps_bins])
ax_eps.set_xlabel('Epsilon')
ax_eps.set_ylabel('Contribution (%)')
ax_eps.set_title('(d) Epsilon distribution', fontsize=10)
ax_eps.grid(True, axis='y', ls=':', lw=0.4, alpha=0.6)

            f'peak = {eps_bins[iE_peak]:.2f}\nmean = {eps_mean:.2f}',

for ax in [ax_M, ax_R, ax_eps]:
    ax.tick_params(direction='in', which='both', top=True, right=True)

disaggregation.mMATLAB
% =============================================================
% 地震危险性解聚 | M-R-epsilon 联合贡献
% -------------------------------------------------------------
% 输入数据：disaggregation.csv
%   M_center,R_center,epsilon,contribution
%   ...
%   单位：M 无量纲, R km, epsilon 无量纲, contribution %
% -------------------------------------------------------------
% 输出：
%   disaggregation.png      2D 四面板
%   disaggregation-3d.png   3D 立体柱状图
% =============================================================

clear; clc; close all;

CSV_FILE = 'disaggregation.csv';

% =============================================================
% 1. 读取与聚合
% =============================================================
if isfile(CSV_FILE)
    T = readtable(CSV_FILE, ...
        'CommentStyle', '#', ...
        'VariableNamingRule', 'preserve');
    M_all   = T.M_center;
    R_all   = T.R_center;
    eps_all = T.epsilon;
    c_all   = T.contribution;
    fprintf('从 %s 读取 %d 个分箱\n', CSV_FILE, numel(c_all));
else
    fprintf('%s 不存在，使用合成数据演示\n', CSV_FILE);
    Ms  = [5.25; 5.75; 6.25; 6.75; 7.25];
    Rs  = [7.5; 15.0; 25.0; 40.0];
    eps = [-0.5; 0.5; 1.5];
    fM  = [0.20; 0.60; 1.00; 0.85; 0.35];
    fR  = [0.30; 1.00; 0.85; 0.45];
    fE  = [0.20; 0.35; 0.45];

    M_all = []; R_all = []; eps_all = []; c_all = [];
    for i = 1:numel(Ms)
        for j = 1:numel(Rs)
            w = fM(i) * fR(j);
            for k = 1:numel(eps)
                M_all   = [M_all; Ms(i)];    %#ok
                R_all   = [R_all; Rs(j)];    %#ok
                eps_all = [eps_all; eps(k)]; %#ok
                c_all   = [c_all; w*fE(k)];  %#ok
            end
        end
    end
    c_all = c_all / sum(c_all) * 100;
end

if abs(sum(c_all) - 100) > 0.5
    c_all = c_all / sum(c_all) * 100;
end

% 唯一分箱
M_bins   = unique(M_all);
R_bins   = unique(R_all);
eps_bins = unique(eps_all);
nM = numel(M_bins);
nR = numel(R_bins);
nE = numel(eps_bins);

% 联合矩阵
MR_joint   = zeros(nM, nR);
MR_eps_sum = zeros(nM, nR);
for i = 1:nM
    for j = 1:nR
        mask = M_all == M_bins(i) & R_all == R_bins(j);
        MR_joint(i,j)   = sum(c_all(mask));
        MR_eps_sum(i,j) = sum(c_all(mask) .* eps_all(mask));
    end
end
MR_eps_mean = MR_joint;
maskPos = MR_joint > 0;
MR_eps_mean(maskPos) = MR_eps_sum(maskPos) ./ MR_joint(maskPos);

% 边际分布
M_marg   = sum(MR_joint, 2);
R_marg   = sum(MR_joint, 1)';
eps_marg = zeros(nE, 1);
for k = 1:nE
    eps_marg(k) = sum(c_all(eps_all == eps_bins(k)));
end

% 峰值与均值
[~, iM_peak] = max(M_marg);
[~, iR_peak] = max(R_marg);
[~, iE_peak] = max(eps_marg);

M_mean   = sum(M_marg .* M_bins)   / sum(M_marg);
R_mean   = sum(R_marg .* R_bins)   / sum(R_marg);
eps_mean = sum(eps_marg .* eps_bins) / sum(eps_marg);

fprintf('M 峰值 = %.2f, 贡献 %.1f%%\n', M_bins(iM_peak), M_marg(iM_peak));

Notes

方法与说明

  • 3D 视图的四个维度：M、R、贡献、ε。 三维空间只能放三个，所以用柱高 = 贡献。
  • 坐标轴视角：经典 bar3d 视角，X 轴（震级）水平， Y 轴（距离）斜向右后，Z 轴竖直向上。三面墙（后墙、右墙、底面） 带网格。
  • 常见问题：
    • 柱子互相遮挡 → 按 (i + j) 排序绘制，远的先画。
    • 贡献百分比总和不为 100 → CSV 未归一化或分箱不完整。
